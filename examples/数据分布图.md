# 数据分布图

用 numpy 生成样本，matplotlib 画分布。**图表不需要写返回值也会自动显示**，和 Jupyter 的 inline 后端一致。

> 画中文标签要指定含中文字形的字体，否则会显示成方框。macOS 上用 `PingFang SC`。

```python {id=p1}
import numpy as np
import matplotlib.pyplot as plt

# 中文标签需要指定字体，否则 matplotlib 默认字体没有中文字形
plt.rcParams["font.sans-serif"] = ["PingFang SC", "Heiti SC", "Arial Unicode MS"]
plt.rcParams["axes.unicode_minus"] = False

rng = np.random.default_rng(42)
normal = rng.normal(170, 7, 2000)
skewed = rng.gamma(2.2, 1.6, 2000) * 6 + 150

print(f"正态样本  均值 {normal.mean():.1f}  标准差 {normal.std():.1f}")
print(f"偏态样本  均值 {skewed.mean():.1f}  中位数 {np.median(skewed):.1f}")
```

## 直方图与核密度

两组样本叠在一起看形状差异。

```python {id=p3}
fig, ax = plt.subplots(figsize=(8, 4.2))

ax.hist(normal, bins=45, density=True, alpha=0.55, color="#3f5a7d", label="正态 μ=170 σ=7")
ax.hist(skewed, bins=45, density=True, alpha=0.55, color="#9c3423", label="偏态 gamma")

for data, color in ((normal, "#2b3f57"), (skewed, "#6e2418")):
    xs = np.linspace(data.min(), data.max(), 300)
    # 高斯核密度估计，避免依赖 scipy
    h = 1.06 * data.std() * len(data) ** (-1 / 5)
    kde = np.exp(-0.5 * ((xs[:, None] - data) / h) ** 2).sum(axis=1) / (len(data) * h * np.sqrt(2 * np.pi))
    ax.plot(xs, kde, color=color, linewidth=1.6)

ax.axvline(normal.mean(), color="#2b3f57", linestyle="--", linewidth=1)
ax.axvline(np.median(skewed), color="#6e2418", linestyle="--", linewidth=1)
ax.set_xlabel("身高 (cm)")
ax.set_ylabel("概率密度")
ax.set_title("两组样本的分布对比")
ax.legend(frameon=False)
ax.spines[["top", "right"]].set_visible(False)
fig.tight_layout()
```

## 箱线图

换个角度看四分位和离群点。

```python {id=p5}
fig, ax = plt.subplots(figsize=(7, 3))
bp = ax.boxplot([normal, skewed], orientation="horizontal", patch_artist=True,
                tick_labels=["正态", "偏态"], widths=0.55)
for patch, color in zip(bp["boxes"], ["#3f5a7d", "#9c3423"]):
    patch.set_facecolor(color)
    patch.set_alpha(0.55)
for median in bp["medians"]:
    median.set_color("#1c1c1a")
ax.set_xlabel("身高 (cm)")
ax.spines[["top", "right"]].set_visible(False)
fig.tight_layout()
```
